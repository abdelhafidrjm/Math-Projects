\section{1st Order Linear ODE}

\begin{align*}
(E)  &\  \dots \ \ \ a(x)y' + b(x)y= f(x)\\
y    &= y_h + y_p\\
y_h  &= Ce^{-\int{\frac{b(x)}{a(x)}}dx}\\
y_p  &= C(x)e^{-\int{\frac{b(x)}{a(x)}}dx} \\
C(x) &= \int{\frac{f(x)}{a(x)}e^{\int{\frac{b(x)}{a(x)}}dx}dx}
\end{align*}

\section{1st Order Nonlinear ODE}

\subsection{Bernoulli}
\begin{align*}
(E) &\iff a(x)y' + b(x)y= y^n f(x) , n > 1\\
(E) &\iff a(x)y'y^{-n} + b(x)y^{-n} = f(x) \\
z   &= y^{1-n} \implies z'= (1-n)y^{-n}y'\\
(E) &\iff a(x)z + (1-n)b(x)z= (1-n)f(x)
\end{align*}

\subsection{Riccati}

\begin{align*}
(E) &\iff a(x)y' + b(x)y= y^2f(x) + g(x) \\ \\
\text{assume }&y_{p} \text{ a particular solution of }(E)\\ 
\implies& y = y_{p} + \frac{1}{z} \text{ with z the solution of the homogenuous equation }(E_{h}) \text{ such that }z = \frac{1}{y}\\ \\
(E_{h}) &\iff a(x)z' - b(x)z= -f(x)
\end{align*}

\section{2nd Order Linear ODE}
$(E) \iff ay''+ by' + cy= f(x)$\\
First, we find its homogenuous equation: $(E_{h}) \iff ay''+ by' + cy= 0$\\
Then,  we find the characteristic equation: $(E_{c}) \iff ar^2 + br + c = 0$\\
We, then have to compute $\Delta = b^2 - 4ac$\\
\begin{itemize}
    \item $\text{If }\Delta > 0, \text{have two solutions: }r_{1}, r_{2}
    \implies y_{h} = Ae^{r_{1}x} + Be^{r_{2}x} |
    \begin{cases} 
        y_{1} &= e^{r_{1}x}  \\
        y_{2} &= e^{r_{2}x}
    \end{cases}$ \\
    \item $\text{If }\Delta = 0, \text{have double solution: }r_0
    \implies y_{h} = (A+Bx)e^{r_{0}x} | 
    \begin{cases} 
        y_{1} &= e^{r_{1}x}  \\
        y_{2} &= xe^{r_{2}x}
    \end{cases}$ \\
    \item $\text{If }\Delta < 0, \text{have two complex solutions: }r = \alpha \pm i\beta 
    \implies y_{h} = e^{\alpha x}(A\cos{\beta x} + B\sin{\beta x}) |
    \begin{cases} 
    y_{1} &= e^{\alpha x}\cos{\beta x}  \\
    y_{2} &= e^{\alpha x}\sin{\beta x}
    \end{cases}$
\end{itemize}

Now, we need to find the particular solution using Lagrange (variation of constants): \\
$y_{p} = A(x) y_{1}+ B(x) y_{2}$ \\
To find $A(x)$ and $B(x)$ we solve the following linear system using carmer:
\[
    \begin{cases}
        A'(x)y_{1} + B'(x)y_{2} = 0 \\
        A'(x)y_{1}' + B'(x)y_{2}' = \frac{f(x)}{a}
    \end{cases} \iff 
    \begin{pmatrix}
        y_{1} & y_{2}\\
        y_{1}' & y_{2}' 
    \end{pmatrix}\begin{pmatrix}
        A'(x)\\
        B'(x)
        \end{pmatrix} = \begin{pmatrix}
        0 \\
        \frac{f(x)}{a}
        \end{pmatrix}
\]

\[
    \implies 
    \begin{cases}
        A'(x) &= \frac{
            \begin{vmatrix}
                0 & y_{2}\\
                \frac{f(x)}{a} & y_{2}' 
            \end{vmatrix}}{
            \begin{vmatrix}
                y_{1} & y_{2}\\
                y_{1}' & y_{2}' 
            \end{vmatrix}}\\
        B'(x) &= \frac{
            \begin{vmatrix}
                y_{1} & 0\\
                y_{1}'  & \frac{f(x)}{a}'
            \end{vmatrix}}{
            \begin{vmatrix}
                y_{1} & y_{2}\\
                y_{1}' & y_{2}' 
            \end{vmatrix}}
    \end{cases}
\]

Then,  we integrate them and replace them in $y_{p}$ \\
Then, the solution of $(E)$ is $y = y_{h}+y_{p}$ \\
Look up: Algebra 02: Carmer's \\

\subsection{Cases of $f(x)$}

If $f(x) = P_{n}e^{\alpha x} | P_{n} \in \mathbb{R}_{n}[X]$, Then the solution is


\begin{itemize}
    \item if $\alpha$ is a simple solution of $(E_{c})$ then $y = xQ_{n}e^{\alpha x}$ \\
    \item if $\alpha$ is a double solution of $(E_{c})$ then $y = x^2Q_{n}e^{\alpha x}$ \\
    \item if $\alpha$ is not a solution of $(E_{c})$ then $y = Q_{n}e^{\alpha x}$
\end{itemize}

If $f(x) = e^{\alpha x} (P_{1}\cos{\beta x}+P_{2}\sin{\beta x}) | P_{1}, P_{2} \in \mathbb{R}_{n}[X]$, Then the solution is 
\begin{itemize}
    \item if $\alpha +i\beta$ is a solution of $(E_{c}) \implies y = xe^{\alpha x} (Q_{1}\cos{\beta x}+Q_{2}\sin{\beta x})$ \\
    \item if $\alpha+i\beta$ is not a solution of $(E_{c}) \implies y = e^{\alpha x} (Q_{1}\cos{\beta x}+Q_{2}\sin{\beta x})$ \\
\end{itemize}

\subsection{Remark}

If \\
\((E) \iff ay'' + by' + cy = \sum_{k=1}^n{f_{k}} = f_{1}(x) + f_{2}(x) + \cdots + f_{n}(x)\) \\
or \\
\((E) \iff a(x)y' + b(x)y = \sum_{k=1}^n{f_{k}}\)

Then we first find the homogenuous solution as usual
then we find the particular solution of the equations $y_{p_{k}}$\\
$ay'' + by' + cy = f_{k}(x)$\\
Or\\
$a(x)y' + b(x)y = f_{k}(x)$ \\
using the previous methods. Then, the solution is: $y = y_{h} + \sum_{k=1}^n  y_{p_{k}}$

\subsection{Remark: Cauchy}

Cauchy aims to find one solution to the differential equation i.e replacing the constants of $y_{h}$ with actual numbers

If the ODE is of 1st order then the conditon given is: $y(x_{0}) = y_{0}$ \\
If the ODE is of 1st order then the conditon given is: $\begin{cases}
y(x_{0}) = y_{0} \\
y'(x_{0}) = y'_{0}
\end{cases}$
